# Edit Mode Implementation Plan

> **For agentic workers:** REQUIRED: Use superpowers:subagent-driven-development (if subagents available) or superpowers:executing-plans to implement this plan. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a persistent edit mode to dev-tools that enables click-to-edit text, image hover bars, and auto-import of non-slot images into airo-media.json.

**Tech Stack:** React (dev-tools), Next.js (app), Express (agents), Vitest (testing), TypeScript throughout.

```typescript
/**
 * Derive a media slot path from dev context.
 * Rules:
 * 1. Extract page from fileName: src/frontend/pages/about.page.tsx → pages/about
 * 2. Kebab-case the component name: HeroSection → hero-section
 * 3. Combine: pages/about/hero-section
 * 4. Fallback: imported/image-{timestamp}
 */
function deriveSlotPath(devContext?: {
  fileName: string;
  componentName: string;
  lineNumber: number;
}): string {
  if (!devContext || !devContext.fileName || !devContext.componentName) {
    return `imported/image-${Date.now()}`;
  }

  // Extract page segment from file path
  // Match patterns like: src/frontend/pages/about.page.tsx or pages/services/web-design.page.tsx
  const pageMatch = devContext.fileName.match(/pages\/(.+?)\.page\.[tj]sx?$/);
  const pageSegment = pageMatch ? `pages/${pageMatch[1]}` : "pages/unknown";

  // Kebab-case the component name: HeroSection → hero-section
  const componentSegment = devContext.componentName
    .replace(/([a-z])([A-Z])/g, "$1-$2")
    .replace(/([A-Z])([A-Z][a-z])/g, "$1-$2")
    .toLowerCase();

  return `${pageSegment}/${componentSegment}`;
}
```

```typescript
  /**
   * Import an arbitrary image URL as a new media slot.
   * Creates a slot in airo-media.json with the URL as currentUrl and first alternative.
   * Derives slot path from dev context (page + component name).
   */
  async importSlot(
    appId: string,
    options: {
      imageUrl: string;
      devContext?: { fileName: string; componentName: string; lineNumber: number };
    }
  ): Promise<{ success: boolean; slotPath?: string; error?: string }> {
    const release = await acquireLock(appId);

    try {
      const result = await this.getManifest(appId);
      if (!result.success || !result.manifest) {
        return { success: false, error: result.error || 'Failed to read manifest' };
      }

      const manifest = result.manifest;

      // Derive base slot path from dev context
      let slotPath = deriveSlotPath(options.devContext);

      // Handle duplicates by appending counter
      if (manifest[slotPath]) {
        let counter = 2;
        while (manifest[`${slotPath}-${counter}`]) {
          counter++;
        }
        slotPath = `${slotPath}-${counter}`;
      }

      // Create the new slot
      const now = new Date().toISOString();
      manifest[slotPath] = {
        mediaType: 'image',
        currentUrl: options.imageUrl,
        searchQuery: '',
        alternatives: [options.imageUrl],
        userModified: false,
        createdAt: now,
        lastUpdated: now,
      };

      // Write manifest atomically (same pattern as updateSlot)
      const app = this.appRepository.findById(appId);
      if (!app?.path) {
        return { success: false, error: 'App not found' };
      }

      const manifestPath = this.getManifestPath(app.path);
      const tmpPath = `${manifestPath}.tmp.${Date.now()}`;
      await fs.writeFile(tmpPath, JSON.stringify(manifest, null, 2));
      await fs.rename(tmpPath, manifestPath);

      logger.info('media_manifest.import_slot.success', { appId, slotPath, imageUrl: options.imageUrl });

      return { success: true, slotPath };
    } catch (error) {
      logger.error('media_manifest.import_slot.error', {
        appId,
        error: error instanceof Error ? error.message : String(error),
      });
      return { success: false, error: 'Failed to import slot' };
    } finally {
      release();
    }
  }
```

```typescript
/**
 * POST /media/:appId/slots/import
 * Auto-import an arbitrary image URL as a new media slot in airo-media.json.
 * Creates the slot with the URL as currentUrl, derives slot path from devContext.
 */
router.post("/:appId/slots/import", async (req: Request, res: Response) => {
  const { appId } = req.params;
  const { imageUrl, devContext } = req.body;

  if (!imageUrl || typeof imageUrl !== "string") {
    res.status(400).json({
      success: false,
      error: "imageUrl is required and must be a string",
    });
    return;
  }

  logger.info("route.media.import_slot.begin", {
    appId,
    imageUrl: imageUrl.substring(0, 200),
  });

  try {
    const result = await mediaManifestService.importSlot(appId, {
      imageUrl,
      devContext,
    });

    if (!result.success) {
      logger.error("route.media.import_slot.failed", {
        appId,
        error: result.error,
      });
      res.status(500).json({
        success: false,
        error: result.error || "Failed to import slot",
      });
      return;
    }

    logger.info("route.media.import_slot.success", {
      appId,
      slotPath: result.slotPath,
    });

    res.json({
      success: true,
      slotPath: result.slotPath,
    });
  } catch (error) {
    logger.error("route.media.import_slot.error", {
      appId,
      error: error instanceof Error ? error.message : String(error),
    });
    res.status(500).json({
      success: false,
      error: "Internal server error",
    });
  }
});
```

```typescript
/** Check if an element is suitable for inline text editing */
function isTextEditable(element: HTMLElement): boolean {
  const tagName = element.tagName.toLowerCase();
  const textContent = element.textContent?.trim() || "";

  const textTags = [
    "p",
    "h1",
    "h2",
    "h3",
    "h4",
    "h5",
    "h6",
    "span",
    "a",
    "button",
    "label",
    "li",
  ];
  const hasText = textContent.length > 0;
  const hasOnlyText =
    element.children.length === 0 ||
    Array.from(element.children).every((child) => {
      const tag = child.tagName.toLowerCase();
      return (
        tag === "br" ||
        tag === "span" ||
        tag === "strong" ||
        tag === "em" ||
        tag === "b" ||
        tag === "i"
      );
    });

  return textTags.includes(tagName) && hasText && hasOnlyText;
}

/** Generate a precise nth-child CSS selector for an element */
function generatePreciseSelector(element: HTMLElement): string {
  const path: string[] = [];
  let current: HTMLElement | null = element;

  while (
    current &&
    current !== document.body &&
    current !== document.documentElement
  ) {
    let selector = current.tagName.toLowerCase();
    if (current.parentElement) {
      const siblings = Array.from(current.parentElement.children);
      const index = siblings.indexOf(current) + 1;
      selector += `:nth-child(${index})`;
    }
    path.unshift(selector);
    current = current.parentElement;
  }

  if (current === document.documentElement) {
    path.unshift("html:nth-child(1)");
  }

  return path.join(" > ");
}

/** Generate a basic CSS selector for an element */
function generateSelector(element: HTMLElement): string {
  if (element.id) return `#${element.id}`;

  let selector = element.tagName.toLowerCase();
  if (element.className && typeof element.className === "string") {
    const classes = element.className
      .split(" ")
      .filter((c) => c.trim() && !c.includes(":"))
      .map((c) => c.replace(/[^\w-]/g, ""))
      .filter((c) => c.length > 0);
    if (classes.length > 0) {
      selector += "." + classes.join(".");
    }
  }
  return selector;
}

/** Extract dev context from element or its ancestors */
function extractDevContext(element: HTMLElement): DevContext | undefined {
  let current: HTMLElement | null = element;
  while (current && current !== document.body) {
    const file = current.getAttribute("data-dev-file");
    const component = current.getAttribute("data-dev-component");
    const line = current.getAttribute("data-dev-line");
    if (file && component && line) {
      return {
        fileName: file
          .replace(/.*\/src\//, "src/")
          .replace(/.*\/app\//, "app/"),
        componentName: component,
        lineNumber: parseInt(line, 10),
      };
    }
    current = current.parentElement;
  }
  return undefined;
}

/** Check if an element is an image or contains/wraps an image */
function detectImage(element: HTMLElement): {
  isImage: boolean;
  imageUrl: string | null;
  type: "img" | "background" | null;
} {
  // Direct <img>
  if (element.tagName.toLowerCase() === "img") {
    const img = element as HTMLImageElement;
    return {
      isImage: true,
      imageUrl: img.src || img.currentSrc || null,
      type: "img",
    };
  }

  // Background image on element
  const bg = window.getComputedStyle(element).backgroundImage;
  if (bg && bg !== "none") {
    const match = bg.match(/url\(["']?([^"')]+)["']?\)/);
    if (match?.[1])
      return { isImage: true, imageUrl: match[1], type: "background" };
  }

  // Contains <img> child
  const imgChild = element.querySelector("img") as HTMLImageElement;
  if (imgChild) {
    return {
      isImage: true,
      imageUrl: imgChild.src || imgChild.currentSrc || null,
      type: "img",
    };
  }

  return { isImage: false, imageUrl: null, type: null };
}

/** Check if a URL is a media slot and extract the slot path */
function getMediaSlotPath(url: string | null): string | null {
  if (!url) return null;
  const match = url.match(/\/airo-assets\/images\/(.+?)(?:\?|$)/);
  return match ? match[1] : null;
}

export function useEditMode(isEditModeActive: boolean) {
  const [state, setState] = useState<EditModeState>({
    editingElement: null,
    originalContent: null,
    originalText: null,
    saveStatus: "idle",
    hoveredImage: null,
  });

  const debounceTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
  const hideBarTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
  const stateRef = useRef(state);
  stateRef.current = state;

  // ── Save logic ──

  const saveCurrentEdit = useCallback(
    (element: HTMLElement, originalText: string) => {
      const newText = element.textContent?.trim() || "";
      if (newText === originalText) return; // No change

      const devContext = extractDevContext(element);
      const selector = generateSelector(element);
      const preciseSelector = generatePreciseSelector(element);

      setState((prev) => ({ ...prev, saveStatus: "saving" }));

      safePostMessage(window.parent, {
        type: "TEXT_UPDATED",
        data: {
          selector,
          preciseSelector,
          oldText: originalText,
          newText,
          devContext,
        },
      });

      showSaveIndicator(element);
      setState((prev) => ({ ...prev, saveStatus: "saved" }));
      setTimeout(() => {
        setState((prev) =>
          prev.saveStatus === "saved" ? { ...prev, saveStatus: "idle" } : prev,
        );
      }, SAVE_INDICATOR_MS);
    },
    [],
  );

  // ── Stop editing ──

  const stopEditing = useCallback(
    (save: boolean) => {
      const { editingElement, originalText, originalContent } =
        stateRef.current;
      if (!editingElement) return;

      // Clear debounce timer
      if (debounceTimerRef.current) {
        clearTimeout(debounceTimerRef.current);
        debounceTimerRef.current = null;
      }

      if (save && originalText !== null) {
        saveCurrentEdit(editingElement, originalText);
      } else if (!save && originalContent !== null) {
        // Revert on Escape
        editingElement.innerHTML = originalContent;
      }

      editingElement.contentEditable = "false";
      editingElement.style.outline = "";
      editingElement.style.outlineOffset = "";

      setState((prev) => ({
        ...prev,
        editingElement: null,
        originalContent: null,
        originalText: null,
      }));
    },
    [saveCurrentEdit],
  );

  // ── Start editing ──

  const startEditing = useCallback(
    (element: HTMLElement) => {
      // Save any current edit first
      if (
        stateRef.current.editingElement &&
        stateRef.current.editingElement !== element
      ) {
        stopEditing(true);
      }

      element.contentEditable = "true";
      element.style.outline = "1px dashed #3b82f6";
      element.style.outlineOffset = "2px";
      element.focus();

      // Place cursor at end
      const range = document.createRange();
      const sel = window.getSelection();
      range.selectNodeContents(element);
      range.collapse(false);
      sel?.removeAllRanges();
      sel?.addRange(range);

      setState((prev) => ({
        ...prev,
        editingElement: element,
        originalContent: element.innerHTML,
        originalText: element.textContent?.trim() || "",
      }));
    },
    [stopEditing],
  );

  // ── Click handler ──

  useEffect(() => {
    if (!isEditModeActive) return;

    const handleClick = (e: MouseEvent) => {
      const target = e.target as HTMLElement;
      if (!target || isDevToolsElement(target)) return;

      // If clicking on the currently editing element, let it handle normally
      if (stateRef.current.editingElement?.contains(target)) return;

      // Check if target is text-editable
      if (isTextEditable(target)) {
        e.preventDefault();
        e.stopPropagation();
        startEditing(target);
        return;
      }

      // Clicked something non-editable — stop current editing (save via blur)
      if (stateRef.current.editingElement) {
        stopEditing(true);
      }
    };

    // Use capture phase so we get the event before the existing select mode
    document.addEventListener("click", handleClick, true);
    return () => document.removeEventListener("click", handleClick, true);
  }, [isEditModeActive, startEditing, stopEditing]);

  // ── Keyboard handler ──

  useEffect(() => {
    if (!isEditModeActive) return;

    const handleKeyDown = (e: KeyboardEvent) => {
      const { editingElement, originalText } = stateRef.current;
      if (!editingElement) return;

      if (e.key === "Escape") {
        e.preventDefault();
        stopEditing(false); // revert
        return;
      }

      if (e.key === "Enter" && isSingleLineElement(editingElement)) {
        e.preventDefault();
        stopEditing(true); // save
        return;
      }

      // Reset debounce timer on any typing
      if (debounceTimerRef.current) {
        clearTimeout(debounceTimerRef.current);
      }
      debounceTimerRef.current = setTimeout(() => {
        if (
          stateRef.current.editingElement &&
          stateRef.current.originalText !== null
        ) {
          saveCurrentEdit(
            stateRef.current.editingElement,
            stateRef.current.originalText,
          );
          // Update originalText to current text after debounce save
          setState((prev) => ({
            ...prev,
            originalText:
              prev.editingElement?.textContent?.trim() || prev.originalText,
          }));
        }
      }, DEBOUNCE_SAVE_MS);
    };

    document.addEventListener("keydown", handleKeyDown, true);
    return () => document.removeEventListener("keydown", handleKeyDown, true);
  }, [isEditModeActive, stopEditing, saveCurrentEdit]);

  // ── Image hover detection ──

  useEffect(() => {
    if (!isEditModeActive) return;

    const handleMouseOver = (e: MouseEvent) => {
      const target = e.target as HTMLElement;
      if (!target || isDevToolsElement(target)) return;

      // Don't show hover bar over the element being text-edited
      if (stateRef.current.editingElement?.contains(target)) return;

      const imageInfo = detectImage(target);
      if (!imageInfo.isImage || !imageInfo.imageUrl) return;

      // Clear any pending hide timer
      if (hideBarTimerRef.current) {
        clearTimeout(hideBarTimerRef.current);
        hideBarTimerRef.current = null;
      }

      const slotPath = getMediaSlotPath(imageInfo.imageUrl);

      setState((prev) => ({
        ...prev,
        hoveredImage: {
          element: target,
          imageUrl: imageInfo.imageUrl!,
          isMediaSlot: slotPath !== null,
          slotPath,
        },
      }));
    };

    const handleMouseOut = (e: MouseEvent) => {
      const relatedTarget = e.relatedTarget as HTMLElement | null;

      // If moving to the hover bar, don't hide
      if (relatedTarget?.closest(".edit-mode-hover-bar")) return;

      // Delay hiding to allow cursor to reach the bar
      hideBarTimerRef.current = setTimeout(() => {
        setState((prev) => ({ ...prev, hoveredImage: null }));
      }, 300);
    };

    document.addEventListener("mouseover", handleMouseOver);
    document.addEventListener("mouseout", handleMouseOut);
    return () => {
      document.removeEventListener("mouseover", handleMouseOver);
      document.removeEventListener("mouseout", handleMouseOut);
      if (hideBarTimerRef.current) clearTimeout(hideBarTimerRef.current);
    };
  }, [isEditModeActive]);

  // ── Cleanup on deactivation ──

  useEffect(() => {
    if (!isEditModeActive && stateRef.current.editingElement) {
      stopEditing(true); // save before deactivating
    }
  }, [isEditModeActive, stopEditing]);

  // ── Hover bar mouse handlers (keep bar visible while cursor is over it) ──

  const handleBarMouseEnter = useCallback(() => {
    if (hideBarTimerRef.current) {
      clearTimeout(hideBarTimerRef.current);
      hideBarTimerRef.current = null;
    }
  }, []);

  const handleBarMouseLeave = useCallback(() => {
    hideBarTimerRef.current = setTimeout(() => {
      setState((prev) => ({ ...prev, hoveredImage: null }));
    }, 300);
  }, []);

  return {
    isEditModeActive,
    editingElement: state.editingElement,
    saveStatus: state.saveStatus,
    hoveredImage: state.hoveredImage,
    handleBarMouseEnter,
    handleBarMouseLeave,
    stopEditing,
  };
}
```

```tsx
/** Generate a precise nth-child CSS selector */
function generatePreciseSelector(element: HTMLElement): string {
  const path: string[] = [];
  let current: HTMLElement | null = element;

  while (
    current &&
    current !== document.body &&
    current !== document.documentElement
  ) {
    let selector = current.tagName.toLowerCase();
    if (current.parentElement) {
      const siblings = Array.from(current.parentElement.children);
      const index = siblings.indexOf(current) + 1;
      selector += `:nth-child(${index})`;
    }
    path.unshift(selector);
    current = current.parentElement;
  }

  if (current === document.documentElement) {
    path.unshift("html:nth-child(1)");
  }

  return path.join(" > ");
}
```

```typescript
case 'AUTO_IMPORT_MEDIA_SLOT': {
  const { imageUrl, devContext } = event.data
  if (!imageUrl || !appId) break

  // Use an IIFE since handleMessage is not async
  ;(async () => {
    logger.info('develop-page.auto-import-media-slot.begin', { appId, imageUrl: imageUrl.substring(0, 200) })

    try {
      // Call agents API directly (same pattern as useMediaSlot hook)
      const tokenResult = await regenerateToken({ siteId: appId, force: true })
      const apiConfig = await dynamicApiConfigService.getConfigForSite(appId)

      if (!apiConfig.agentUrl) {
        throw new Error('Agent URL not available')
      }

      const response = await fetch(
        `${apiConfig.agentUrl}/media/${appId}/slots/import`,
        {
          method: 'POST',
          headers: {
            'Authorization': `Bearer ${tokenResult.token}`,
            'Content-Type': 'application/json',
          },
          body: JSON.stringify({ imageUrl, devContext }),
        }
      )

      if (!response.ok) {
        throw new Error(`Import failed: ${response.status}`)
      }

      const result = await response.json()
      if (result.success && result.slotPath) {
        // Open media dialog with the new slot
        useMediaSlotDialogStore.getState().openDialog(appId, result.slotPath)
      }
    } catch (error) {
      logger.error('develop-page.auto-import-media-slot.error', {
        appId,
        error: error instanceof Error ? error.message : String(error),
      })
      // TODO: Show error toast — integrate with existing toast system if available
    }
  })()
  break
}
```
